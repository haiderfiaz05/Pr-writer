````markdown
## 🔍 Notable diffs

-async function charge(api, amount) {
-  return api.post('/charge', { amount });
-}
+async function charge(api, amount, { retries = 3, baseDelayMs = 200 } = {}) {
+  for (let attempt = 0; ; attempt++) {
+    try {
+      return await api.post('/charge', { amount });
+    } catch (err) {
+      if (err.status !== 503 || attempt >= retries) throw err;
+      await sleep(baseDelayMs * 2 ** attempt);
+    }
+  }
+}
```
````
